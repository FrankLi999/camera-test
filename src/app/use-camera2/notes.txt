mobile portrait: widthRation: 0.5, heightRation: 0.5
landscape: widthRatio: 0.25, heightRation: 0.65 

overlayWidth: canvas.width * widthRatio
overlayHeight: canvas.height * heightRatio

overlayX = (canvas.width - overlayWidth) /2

overlayY = (canvas.height - overlayHeight) / 2

eclipse(
    overlayX + overlayWidth /2, overlayY + overlayHeight /2,
    overlayWidth /2, overlayHeight /2,
    0, 0, # rotation/startAngle
    2 * Math.PI # endAngle
)
